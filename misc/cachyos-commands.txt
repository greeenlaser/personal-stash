#add ssh for git + github
1) ssh-keygen -t ed25519 -C "username@github" #change username to github username
2) sudo pacman -S ksshaskpass
3) set -gx SSH_ASKPASS /usr/bin/ksshaskpass
4) set -gx SSH_AUTH_SOCK $XDG_RUNTIME_DIR/ssh-agent.socket
5) systemctl --user enable --now ssh-agent.service
6) ssh-add ~/.ssh/id_ed25519
7) cat ~/.ssh/id_ed25519.pub #copy full value to github ssh and gpg keys -> new ssh key -> key field
8) ssh -T git@github.com
